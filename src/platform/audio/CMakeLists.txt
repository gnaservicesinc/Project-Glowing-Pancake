option(PGP_ENABLE_AUDIO "Compile optional native playback; silent fallback always remains" ${APPLE})

set(pgp_audio_vendor "${PROJECT_SOURCE_DIR}/third_party/audio")
add_library(pgp_audio_vorbis STATIC stb_vorbis_impl.cpp)
target_include_directories(pgp_audio_vorbis SYSTEM PRIVATE "${pgp_audio_vendor}/stb")
target_compile_definitions(pgp_audio_vorbis PRIVATE STB_VORBIS_NO_STDIO STB_VORBIS_NO_PUSHDATA_API)

add_library(pgp_audio STATIC player.cpp clip.cpp)
add_library(pgp::audio ALIAS pgp_audio)
target_include_directories(pgp_audio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(pgp_audio SYSTEM PRIVATE "${pgp_audio_vendor}/stb")
target_compile_definitions(pgp_audio PUBLIC PGP_ENABLE_AUDIO=$<BOOL:${PGP_ENABLE_AUDIO}>)
target_compile_definitions(pgp_audio PRIVATE STB_VORBIS_NO_STDIO STB_VORBIS_NO_PUSHDATA_API)
target_link_libraries(pgp_audio PRIVATE pgp_audio_vorbis)
pgp_project_target(pgp_audio)

if(PGP_ENABLE_AUDIO)
    add_library(pgp_audio_native STATIC miniaudio_impl.cpp)
    target_include_directories(pgp_audio_native SYSTEM PRIVATE "${pgp_audio_vendor}/miniaudio")
    target_include_directories(pgp_audio SYSTEM PRIVATE "${pgp_audio_vendor}/miniaudio")
    target_link_libraries(pgp_audio PRIVATE pgp_audio_native)
    find_package(Threads REQUIRED)
    target_link_libraries(pgp_audio_native PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    if(APPLE)
        target_link_libraries(pgp_audio_native PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
    elseif(WIN32)
        target_link_libraries(pgp_audio_native PRIVATE ole32)
    endif()
endif()

if(BUILD_TESTING)
    add_subdirectory("${PROJECT_SOURCE_DIR}/tests/audio" "${CMAKE_CURRENT_BINARY_DIR}/tests")
endif()
